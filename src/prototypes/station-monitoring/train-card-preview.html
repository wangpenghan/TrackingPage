import React from 'react';

const TrainCard = ({ trainData, design = 1 }) => {
  const designs = {
    1: { primary: '#1e40af', secondary: '#3b82f6', accent: '#0ea5e9' },
    2: { primary: '#0f766e', secondary: '#14b8a6', accent: '#5eead4' },
    3: { primary: '#581c87', secondary: '#7c3aed', accent: '#a78bfa' },
    4: { primary: '#312e81', secondary: '#4338ca', accent: '#8b5cf6' },
    5: { primary: '#27272a', secondary: '#3f3f46', accent: '#52525b' },
  };
  
  const colors = designs[design];
  
  return (
    <div className="card-container">
      {/* SVG车头剪影 */}
      <svg viewBox="0 0 288 80">
        <defs>
          <linearGradient id="trainGrad" x1="0%" y1="0%" x2="0%" y2="100%">
            <stop offset="0%" stopColor={colors.primary} />
            <stop offset="100%" stopColor={colors.secondary} />
          </linearGradient>
        </defs>
        {/* 车头路径... */}
      </svg>
      
      {/* 信息内容 */}
      <div className="p-3 pt-20">
        <span style={{ color: colors.primary }}>
          {trainData.trainNo}
        </span>
        <span style={{ color: colors.secondary }}>
          {trainData.arrivalTime}
        </span>
      </div>
    </div>
  );
};

export default TrainCard;
